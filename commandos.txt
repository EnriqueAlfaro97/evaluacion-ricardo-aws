# ============================================================================== 
# 1. VARIABLES DE ENTORNO
# ==============================================================================
export AWS_REGION=us-east-1
export CLUSTER=evaluacion-ricardo
export ECR_REPO=evaluacion-ricardo
export GH_REPO=EnriqueAlfaro97/evaluacion-ricardo-aws
export ACCOUNT_ID=$(aws sts get-caller-identity --query Account --output text)
export REGISTRY=$ACCOUNT_ID.dkr.ecr.$AWS_REGION.amazonaws.com

echo "Cuenta: $ACCOUNT_ID | Región: $AWS_REGION | Repo: $GH_REPO"


# ==============================================================================
# 2. CREACIÓN DE CLÚSTER EKS
# Asegúrate de que cluster.yaml tenga name: evaluacion-ricardo y region: us-east-1
# ==============================================================================
eksctl create cluster -f cluster.yaml
kubectl get nodes -o wide
kubectl get pods -n kube-system


# ==============================================================================
# 3. CREACIÓN DE ECR, CONSTRUCCIÓN Y SUBIDA DE IMAGEN (Versión Inicial v1)
# ==============================================================================
# Crear el repositorio en ECR (si no existe)
aws ecr create-repository \
  --repository-name $ECR_REPO \
  --image-scanning-configuration scanOnPush=true \
  --region $AWS_REGION 2>/dev/null || echo "El repositorio ECR ya existe"

aws ecr get-login-password --region $AWS_REGION | docker login --username AWS --password-stdin $REGISTRY

docker build -t $REGISTRY/$ECR_REPO:v1 .
docker push $REGISTRY/$ECR_REPO:v1

aws ecr list-images --repository-name $ECR_REPO --region $AWS_REGION \
  --query 'imageIds[].imageTag' --output table

aws ecr describe-images --repository-name $ECR_REPO --region $AWS_REGION \
  --query 'sort_by(imageDetails,&imagePushedAt)[].{Tag:imageTags[0],Bytes:imageSizeInBytes,Subida:imagePushedAt}' \
  --output table


# ==============================================================================
# 4. DESPLIEGUE INICIAL EN KUBERNETES Y PRUEBA LOCAL
# ==============================================================================
kubectl apply -f k8s/namespace.yaml
kubectl apply -f k8s/deployment.yaml -f k8s/service.yaml
kubectl -n evaluacion-ricardo rollout status deployment/evaluacion-ricardo
kubectl -n evaluacion-ricardo get pods -o wide

# Port-forward para probar localmente:
# kubectl -n evaluacion-ricardo port-forward svc/evaluacion-ricardo 8080:80
# curl localhost:8080


# ==============================================================================
# 5. ACTUALIZACIÓN MANUAL DE IMAGEN Y ROLLBACK (Prueba de v2)
# ==============================================================================
docker build -t $REGISTRY/$ECR_REPO:v2 . && docker push $REGISTRY/$ECR_REPO:v2
kubectl -n evaluacion-ricardo set image deployment/evaluacion-ricardo evaluacion-ricardo=$REGISTRY/$ECR_REPO:v2
kubectl -n evaluacion-ricardo rollout status deployment/evaluacion-ricardo

# Rollback en caso de requerir volver a v1:
# kubectl -n evaluacion-ricardo rollout undo deployment/evaluacion-ricardo


# ==============================================================================
# 6. INSTALACIÓN DE AWS LOAD BALANCER CONTROLLER
# ==============================================================================
curl -o alb-iam-policy.json \
  https://raw.githubusercontent.com/kubernetes-sigs/aws-load-balancer-controller/main/docs/install/iam_policy.json

aws iam create-policy \
  --policy-name AWSLoadBalancerControllerIAMPolicy \
  --policy-document file://alb-iam-policy.json

eksctl create iamserviceaccount \
  --cluster=$CLUSTER --region=$AWS_REGION \
  --namespace=kube-system --name=aws-load-balancer-controller \
  --role-name AmazonEKSLoadBalancerControllerRole \
  --attach-policy-arn=arn:aws:iam::$ACCOUNT_ID:policy/AWSLoadBalancerControllerIAMPolicy \
  --approve

helm repo add eks https://aws.github.io/eks-charts && helm repo update

helm upgrade --install aws-load-balancer-controller \
  eks/aws-load-balancer-controller -n kube-system \
  --set clusterName=$CLUSTER \
  --set serviceAccount.create=false \
  --set serviceAccount.name=aws-load-balancer-controller \
  --set region=$AWS_REGION

kubectl -n kube-system rollout status deploy/aws-load-balancer-controller


# ==============================================================================
# 7. INTEGRACIÓN CON GITHUB ACTIONS (OIDC Y ASOCIACIÓN DE POLÍTICAS)
# ==============================================================================
# Crear el proveedor de identidad si no existe aún
aws iam create-open-id-connect-provider \
  --url https://token.actions.githubusercontent.com \
  --client-id-list sts.amazonaws.com 2>/dev/null || echo "OIDC Provider ya existe"

# Reemplazos dinámicos en los archivos de permisos IAM
sed -i "s/[0-9]\{12\}/$ACCOUNT_ID/g" iam/trust-policy.json iam/deploy-policy.json
sed -i "s|repo:[^:]*:|repo:$GH_REPO:|g" iam/trust-policy.json

# Crear el rol de IAM
aws iam create-role --role-name GitHubActionsEKSDeploy \
  --assume-role-policy-document file://iam/trust-policy.json

# Asociar la política de despliegue
aws iam put-role-policy --role-name GitHubActionsEKSDeploy \
  --policy-name deploy --policy-document file://iam/deploy-policy.json

# Obtener ARN del rol y configurar secreto en GitHub
export ROLE_ARN=$(aws iam get-role --role-name GitHubActionsEKSDeploy --query 'Role.Arn' --output text)
echo "$GH_REPO | $ROLE_ARN"
gh secret set AWS_DEPLOY_ROLE_ARN --repo "$GH_REPO" --body "$ROLE_ARN"

# Otorgar acceso al rol dentro del clúster EKS para el namespace 'evaluacion-ricardo'
aws eks create-access-entry \
  --cluster-name $CLUSTER --region $AWS_REGION \
  --principal-arn $ROLE_ARN --type STANDARD

aws eks associate-access-policy \
  --cluster-name $CLUSTER --region $AWS_REGION \
  --principal-arn $ROLE_ARN \
  --policy-arn arn:aws:eks::aws:cluster-access-policy/AmazonEKSEditPolicy \
  --access-scope type=namespace,namespaces=evaluacion-ricardo

# Validar condición configurada en el rol
aws iam get-role --role-name GitHubActionsEKSDeploy \
  --query 'Role.AssumeRolePolicyDocument.Statement[0].Condition'
